\chapter*{符号说明}

\begin{spacing}{1.6}
\begin{center}
\begin{tabular}{r p{0.58\textwidth}}
$a$ & 变量 \\
$\mathbf{a}$ & 行向量或矩阵 \\
$\mathbf{a}^\top$ & 向量或矩阵 $\mathbf{a}$ 的转置 \\
$f(a)$ & $a$ 的函数 \\
$\max f(a)$ & $f(a)$ 的最大值 \\
$\argmax_{a} f(a)$ & 使 $f(a)$ 取得最大值的 $a$ \\
$\mathbf{x}$ & 模型的输入词元序列 \\
$x_j$ & 第 $j$ 个位置的输入词元 \\
$\mathbf{y}$ & 模型生成的输出词元序列 \\
$y_i$ & 第 $i$ 个位置的输出词元 \\
$|\mathbf{x}|$（或 $|\mathbf{y}|$） & 序列 $\mathbf{x}$（或 $\mathbf{y}$）的长度  \\
$V$ & 词表 \\
$\theta$ & 模型参数 \\
$\Pr(a)$ & $a$ 的概率 \\
$\Pr(a|b)$ & 给定 $b$ 时 $a$ 的条件概率 \\
$\Pr(\cdot|b)$ & 给定 $b$ 时某变量的概率分布 \\
$\Pr_{\theta}(a)$ &  由参数 $\theta$ 确定的 $a$ 的概率 \\
$\mathbb{E}_{a\sim \Pr(a)}[f(a)]$ & $a$ 服从分布 $\Pr(a)$ 时 $f(a)$ 的期望值 \\
$\mathbf{h}_t$ & 序列模型在时间步 $t$ 的隐状态 \\
$\mathbf{H}$ & 序列中各时间步隐状态组成的矩阵 \\
$\mathbf{Q}$, $\mathbf{K}$, $\mathbf{V}$ & 注意力机制中的查询、键和值矩阵 \\
$\mathrm{Softmax}(\mathbf{A})$ & 对输入向量或矩阵 $\mathbf{A}$ 进行归一化的 Softmax 函数 \\
$\mathcal{L}$ & 损失函数 \\
$\mathcal{D}$ & 用于训练或微调模型的数据集 \\
$\frac{\partial \mathcal{L}}{\partial \theta}$ & 损失函数 $\mathcal{L}$ 对参数 $\theta$ 的梯度 \\
$D_{\mathrm{KL}}(p\ ||\ q)$ & 分布 $p$ 与 $q$ 之间的 KL 散度
\end{tabular}
\end{center}
\end{spacing}
